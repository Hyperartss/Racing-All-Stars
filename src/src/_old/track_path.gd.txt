class_name TrackPath
extends RefCounted
## Centreline of a race track: a closed Catmull-Rom spline resampled at uniform
## arc-length spacing, with an orthonormal frame (tangent / right / up) per sample.
##
## Frames are what let banking, loops and half-pipes work without special cases:
##   * WORLD  - up is world-up projected across the tangent. Ordinary road.
##             `bank` rolls the surface about the tangent.
##   * RADIAL - up points away from `loop_center`. Vertical loops and pipes, where
##             world-up degenerates at the top of a circle.

const SPACING := 2.0                   ## metres between resampled points
const UP := Vector3.UP

enum Surface { ASPHALT, SAND, ISLAND, CHECKER }
enum Wall { NONE, RAIL, BARRIER, PIPE }


class Sample:
	var idx := 0
	var pos := Vector3.ZERO
	var tangent := Vector3.FORWARD
	var right := Vector3.RIGHT
	var up := Vector3.UP
	var width := 12.0
	var bank := 0.0
	var dist := 0.0
	var loop_center := Vector3.ZERO
	var radial := false
	var surface: int = Surface.ASPHALT
	var wall_l: int = Wall.RAIL
	var wall_r: int = Wall.RAIL
	var flags := 0        ## 1 = start/finish line, 2 = checkpoint gate
	var curv := 0.0       ## signed curvature (rad/m), positive = turning left


var ctrl: Array = []
var samples: Array[Sample] = []
var length := 0.0
var _near_cache := 0


func build(points: Array) -> void:
	ctrl = points.duplicate(true)
	if ctrl.size() < 4:
		push_error("TrackPath.build: needs at least 4 control points")
		return
	var dense := _sample_dense()
	_resample(dense)
	_build_frames()
	_build_curvature()


# --- Catmull-Rom -----------------------------------------------------------

static func _cr(p0: Vector3, p1: Vector3, p2: Vector3, p3: Vector3, t: float) -> Vector3:
	var t2 := t * t
	var t3 := t2 * t
	return 0.5 * ((2.0 * p1) + (p2 - p0) * t
		+ (2.0 * p0 - 5.0 * p1 + 4.0 * p2 - p3) * t2
		+ (-p0 + 3.0 * p1 - 3.0 * p2 + p3) * t3)


static func _blend_val(a: Dictionary, b: Dictionary, key: String, t: float) -> float:
	return lerpf(float(a.get(key, 0.0)), float(b.get(key, 0.0)), t)


func _sample_dense() -> Array[Sample]:
	var out: Array[Sample] = []
	var n := ctrl.size()
	const SUB := 20
	for i in n:
		var a: Dictionary = ctrl[(i - 1 + n) % n]
		var b: Dictionary = ctrl[i]
		var c: Dictionary = ctrl[(i + 1) % n]
		var d: Dictionary = ctrl[(i + 2) % n]
		var ca: Vector3 = a.get("loop_center", Vector3.ZERO)
		var cb: Vector3 = b.get("loop_center", Vector3.ZERO)
		var both_radial := ca != Vector3.ZERO and cb != Vector3.ZERO
		for s in SUB:
			var t := float(s) / float(SUB)
			var sm := Sample.new()
			sm.pos = _cr(a.pos, b.pos, c.pos, d.pos, t)
			sm.width = _blend_val(a, b, "width", t)
			sm.bank = _blend_val(a, b, "bank", t)
			sm.radial = both_radial
			sm.loop_center = ca.lerp(cb, t) if both_radial else Vector3.ZERO
			sm.wall_l = a.wall_l if t < 0.5 else b.wall_l
			sm.wall_r = a.wall_r if t < 0.5 else b.wall_r
			sm.surface = a.surface if t < 0.5 else c.surface
			out.append(sm)
	out[0].flags |= 1
	return out


func _resample(dense: Array[Sample]) -> void:
	var n := dense.size()
	var cum := PackedFloat32Array()
	cum.resize(n + 1)
	var total := 0.0
	for i in n:
		total += dense[i].pos.distance_to(dense[(i + 1) % n].pos)
		cum[i + 1] = total
	length = total

	var count := maxi(4, int(total / SPACING))
	samples.clear()
	for k in count:
		var target := float(k) * SPACING
		var i := _seg_for(cum, target)
		var seg_len := cum[i + 1] - cum[i]
		var t := 0.0 if seg_len < 0.0001 else clampf((target - cum[i]) / seg_len, 0.0, 1.0)
		var a := dense[i]
		var b := dense[(i + 1) % n]
		var sm := Sample.new()
		sm.idx = k
		sm.pos = a.pos.lerp(b.pos, t)
		sm.width = lerpf(a.width, b.width, t)
		sm.bank = lerpf(a.bank, b.bank, t)
		sm.radial = a.radial
		sm.loop_center = a.loop_center.lerp(b.loop_center, t)
		sm.wall_l = a.wall_l
		sm.wall_r = b.wall_r
		sm.surface = a.surface
		sm.dist = target
		sm.flags = a.flags if t < 0.5 else b.flags
		samples.append(sm)


static func _seg_for(cum: PackedFloat32Array, target: float) -> int:
	var lo := 0
	var hi := cum.size() - 2
	while lo < hi:
		var mid := (lo + hi) >> 1
		if cum[mid] <= target:
			lo = mid + 1
		else:
			hi = mid
	return maxi(0, lo - 1)


# --- Frames ----------------------------------------------------------------

func _build_frames() -> void:
	var n := samples.size()
	# Tangents first (central difference over the closed ring).
	for i in n:
		samples[i].tangent = (samples[(i + 1) % n].pos - samples[(i - 1 + n) % n].pos).normalized()

	# Provisional right vectors, then a second pass that can borrow a neighbour's
	# frame wherever the tangent is parallel to the reference up (top of a loop).
	var prov: Array[Vector3] = []
	prov.resize(n)
	for i in n:
		prov[i] = _ref_right(samples[i], i, prov, false)

	for i in n:
		var sm := samples[i]
		var right := _ref_right(sm, i, prov, true)
		sm.right = right
		sm.up = right.cross(sm.tangent).normalized()

	# Apply banking as a roll about the tangent.
	for i in n:
		var sm := samples[i]
		if absf(sm.bank) > 0.00001:
			var b := Basis(sm.tangent, sm.bank)
			sm.right = (b * sm.right).normalized()
			sm.up = (b * sm.up).normalized()

	# Light smoothing so banking transitions do not crease.
	for _pass in 2:
		var sr: Array[Vector3] = []
		var su: Array[Vector3] = []
		for i in n:
			sr.append((samples[(i - 1 + n) % n].right + samples[i].right * 2.0 + samples[(i + 1) % n].right).normalized())
			su.append((samples[(i - 1 + n) % n].up + samples[i].up * 2.0 + samples[(i + 1) % n].up).normalized())
		for i in n:
			samples[i].right = sr[i]
			samples[i].up = su[i]


func _ref_up(sm: Sample) -> Vector3:
	if sm.radial:
		# Away from the loop centre: world-up at the bottom of a loop,
		# world-down at the top. The opposite sign would invert the road.
		var v := sm.loop_center - sm.pos
		if v.length_squared() > 0.0001:
			return v.normalized()
	return UP


func _ref_right(sm: Sample, i: int, prov: Array[Vector3], allow_borrow: bool) -> Vector3:
	var r := sm.tangent.cross(_ref_up(sm))
	if r.length_squared() > 0.00001:
		return r.normalized()
	if allow_borrow:
		# Tangent is parallel to the reference up: reuse a neighbouring frame.
		var n := samples.size()
		for d in range(1, n):
			for sgn in [1, -1]:
				var j := posmod(i + sgn * d, n)
				if prov[j].length_squared() > 0.0001 and samples[j].tangent.dot(sm.tangent) < 0.995:
					return prov[j]
	return Vector3.RIGHT


func _build_curvature() -> void:
	var n := samples.size()
	for i in n:
		var p := samples[(i - 1 + n) % n].tangent
		var q := samples[(i + 1) % n].tangent
		var d := samples[(i + 1) % n].pos.distance_to(samples[(i - 1 + n) % n].pos)
		if d < 0.0001:
			samples[i].curv = 0.0
		else:
			samples[i].curv = signf(p.cross(q).dot(samples[i].up)) * p.angle_to(q) / d


# --- Queries ---------------------------------------------------------------

func sample_index_at_distance(d: float) -> int:
	if samples.is_empty():
		return 0
	return posmod(int(round(fposmod(d, length) / SPACING)), samples.size())


func sample_at_distance(d: float) -> Sample:
	return samples[sample_index_at_distance(d)]


func sample_at_index(i: int) -> Sample:
	return samples[posmod(i, samples.size())]


## Index of the sample closest to a world point, searching a window around the
## last hit first and only scanning everything if that window clearly misses.
func nearest_index(p: Vector3, window: int = 48) -> int:
	var n := samples.size()
	if n == 0:
		return 0
	var lo := maxi(0, _near_cache - window)
	var hi := mini(n - 1, _near_cache + window)
	var best := _near_cache
	var best_d := INF
	for i in range(lo, hi + 1):
		var d := samples[i].pos.distance_squared_to(p)
		if d < best_d:
			best_d = d
			best = i
	if best_d > 40.0 * 40.0:
		for i in n:
			var d := samples[i].pos.distance_squared_to(p)
			if d < best_d:
				best_d = d
				best = i
	_near_cache = best
	return best


## Signed lateral offset of `p` from the centreline at sample `i`, in metres.
## Positive means to the right of travel direction.
func lateral_offset(i: int, p: Vector3) -> float:
	return (p - samples[posmod(i, samples.size())].pos).dot(samples[posmod(i, samples.size())].right)


## Forward distance along the track at the nearest sample to `p`.
func distance_of(p: Vector3) -> float:
	return samples[nearest_index(p)].dist
