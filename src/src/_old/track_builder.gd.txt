class_name TrackBuilder
extends RefCounted
## Turns a TrackPath into renderable geometry: the road ribbon, side
## walls/barriers, and a skirt so the road reads as a solid slab rather than a
## floating plane.
##
## The road is generated as one closed strip around the whole path, so a lap is
## continuous by construction - there is no join between "sections" that could seam.

const LAT_SEGS := 8        ## lateral subdivisions across the road
const TILE := 8.0          ## world metres per texture tile
const WALL_SEGS := 3       ## vertical subdivisions along a wall
const SKIRT_DEPTH := 2.4   ## how far the slab edge drops
const PIPE_RADIUS := 3.6

enum Wall { NONE, RAIL, BARRIER, PIPE }

const WALL_HEIGHT := [0.0, 1.05, 1.35, 4.0]
const WALL_LEAN := [0.0, 0.18, 0.30, 0.0]


## Builds road + walls + skirt.
## Returns {
##   "roads": Dictionary<int, ArrayMesh>  one mesh per TrackPath.Surface used,
##   "walls": ArrayMesh,
##   "road_tris": PackedVector3Array,
##   "wall_tris": PackedVector3Array
## }
static func build(path: TrackPath) -> Dictionary:
	var n := path.samples.size()
	if n < 3:
		return {}

	# Road is emitted per surface type so each band can carry its own texture.
	var used: Dictionary = {}
	for s in path.samples:
		used[s.surface] = true
	var geos: Dictionary = {}
	for k in used:
		geos[k] = _Geometry.new()

	var road_tris := PackedVector3Array()
	for k in geos:
		_build_road(path, geos[k], k)
		road_tris.append_array(geos[k].positions)

	var wall := _Geometry.new()
	var skirt := _Geometry.new()
	_build_walls(path, wall)
	_build_skirt(path, skirt)

	var roads: Dictionary = {}
	for k in geos:
		roads[k] = geos[k].to_mesh()

	var wall_mesh := wall.to_mesh()
	wall_mesh.add_surface_from_arrays(Mesh.PRIMITIVE_TRIANGLES, skirt.to_arrays())

	return {
		"roads": roads,
		"walls": wall_mesh,
		"road_tris": road_tris,
		"wall_tris": wall.positions + skirt.positions,
	}


# --- Road ------------------------------------------------------------------

static func _build_road(path: TrackPath, g: _Geometry, surface: int) -> void:
	var n := path.samples.size()
	var ring := LAT_SEGS + 1
	var base := g.positions.size()

	# Only emit quads whose two samples share the requested surface, so surface
	# bands end cleanly instead of blending across a boundary.
	for i in n:
		var s0 := path.samples[i]
		var s1 := path.samples[(i + 1) % n]
		if s0.surface != surface and s1.surface != surface:
			continue
		var want0 := s0.surface == surface
		var want1 := s1.surface == surface
		var row0 := g.positions.size()
		for j in ring:
			var u := float(j) / float(LAT_SEGS)        # 0 = left edge, 1 = right edge
			var src: TrackPath.Sample = s0 if want0 else s1
			var lat := (u - 0.5) * src.width
			var crown := -absf(lat) * 0.005             # very slight camber
			g.positions.append(src.pos + src.right * lat + src.up * crown)
			g.normals.append(src.up)
			g.uvs.append(Vector2(u * src.width / TILE, src.dist / TILE))
		if not want1:
			continue
		var row1 := g.positions.size()
		for j in ring:
			var u := float(j) / float(LAT_SEGS)
			var lat := (u - 0.5) * s1.width
			var crown := -absf(lat) * 0.005
			g.positions.append(s1.pos + s1.right * lat + s1.up * crown)
			g.normals.append(s1.up)
			g.uvs.append(Vector2(u * s1.width / TILE, s1.dist / TILE))
		for j in LAT_SEGS:
			g.add_quad(row0 + j, row1 + j, row1 + j + 1, row0 + j + 1)


# --- Walls -----------------------------------------------------------------

static func _build_walls(path: TrackPath, g: _Geometry) -> void:
	var n := path.samples.size()
	for side_i in 2:
		var side := 1.0 if side_i == 0 else -1.0
		var kind := _dominant_wall(path, side_i)
		var height: float = WALL_HEIGHT[kind]
		if height <= 0.0:
			continue
		var lean: float = WALL_LEAN[kind]
		var steps := WALL_SEGS + 1
		var base := g.positions.size()

		for i in n:
			var s := path.samples[i]
			var edge := s.pos + s.right * (side * s.width * 0.5)
			var outward := s.right * side
			for k in steps:
				var f := float(k) / float(WALL_SEGS)
				var off := Vector3.ZERO
				if kind == Wall.PIPE:
					# Concave channel: arc out then back in, so a half-pipe forms.
					var ang := f * 1.30
					off = outward * sin(ang) * PIPE_RADIUS \
						+ s.up * ((1.0 - cos(ang)) * PIPE_RADIUS - (1.0 - cos(1.30)) * PIPE_RADIUS + height * f)
				else:
					off = s.up * (height * f) + outward * (lean * height * f * f)
				g.positions.append(edge + off)
				g.normals.append(-outward if kind != Wall.PIPE else (outward * cos(f * 1.30) - s.up * sin(f * 1.30)).normalized())
				g.uvs.append(Vector2(f * height / TILE, s.dist / TILE))

		for i in n:
			var i2 := (i + 1) % n
			var r0 := base + i * steps
			var r1 := base + i2 * steps
			for k in WALL_SEGS:
				var a := r0 + k
				var b := r1 + k
				var c := r1 + k + 1
				var d := r0 + k + 1
				# Wind so the face points away from the road centre.
				if side > 0.0:
					g.add_quad(a, b, c, d)
				else:
					g.add_quad(a, d, c, b)


static func _dominant_wall(path: TrackPath, side_i: int) -> int:
	var tally := {}
	for s in path.samples:
		var k: int = s.wall_r if side_i == 0 else s.wall_l
		tally[k] = int(tally.get(k, 0)) + 1
	var best := 0
	var best_n := -1
	for k in tally:
		if int(tally[k]) > best_n:
			best_n = int(tally[k])
			best = int(k)
	return best


# --- Skirt -----------------------------------------------------------------

static func _build_skirt(path: TrackPath, g: _Geometry) -> void:
	var n := path.samples.size()
	for side_i in 2:
		var side := 1.0 if side_i == 0 else -1.0
		var base := g.positions.size()
		for i in n:
			var s := path.samples[i]
			var e := s.pos + s.right * (side * s.width * 0.5)
			g.positions.append(e)
			g.normals.append(s.right * side)
			g.uvs.append(Vector2(0.0, s.dist / TILE))
			g.positions.append(e - s.up * SKIRT_DEPTH)
			g.normals.append(s.right * side)
			g.uvs.append(Vector2(SKIRT_DEPTH / TILE, s.dist / TILE))
		for i in n:
			var i2 := (i + 1) % n
			var r0 := base + i * 2
			var r1 := base + i2 * 2
			if side > 0.0:
				g.add_quad(r0, r0 + 1, r1 + 1, r1)
			else:
				g.add_quad(r0, r1, r1 + 1, r0 + 1)


# --- Small geometry accumulator -------------------------------------------

class _Geometry:
	var positions := PackedVector3Array()
	var normals := PackedVector3Array()
	var uvs := PackedVector2Array()
	var indices := PackedInt32Array()

	func add_quad(a: int, b: int, c: int, d: int) -> void:
		indices.append_array([a, b, c, a, c, d])

	func to_arrays() -> Array:
		return [positions, normals, uvs, indices]

	func to_mesh() -> ArrayMesh:
		var m := ArrayMesh.new()
		m.add_surface_from_arrays(Mesh.PRIMITIVE_TRIANGLES, to_arrays())
		return m
