extends SceneTree
## Builds the track headlessly and dumps the centreline + a few diagnostics so the
## layout can be checked (and plotted) without opening the editor.

func _init() -> void:
	var pts := TrackLayout.build()
	var path := TrackPath.new()
	path.build(pts)
	print("samples=%d length=%.1f m  control points=%d" % [path.samples.size(), path.length, pts.size()])

	# Surface / wall usage
	var surf := {}
	for s in path.samples:
		surf[s.surface] = int(surf.get(s.surface, 0)) + 1
	print("surface bands (0=ASPHALT 1=SAND 2=ISLAND 3=CHECKER): ", surf)
	var radial := 0
	for s in path.samples:
		if s.radial:
			radial += 1
	print("radial (loop) samples: %d (%.0f m)" % [radial, radial * TrackPath.SPACING])

	# Elevation extremes
	var lo := INF
	var hi := -INF
	for s in path.samples:
		lo = minf(lo, s.pos.y)
		hi = maxf(hi, s.pos.y)
	print("height: min=%.1f max=%.1f" % [lo, hi])

	# Min turn radius (kart-relevant)
	var min_r := INF
	var min_at := 0
	for i in path.samples.size():
		var s := path.samples[i]
		if absf(s.curv) > 0.0001:
			var r := 1.0 / absf(s.curv)
			if r < min_r:
				min_r = r
				min_at = i
	print("tightest corner radius: %.1f m at sample %d (dist %.0f m)" % [min_r, min_at, path.samples[min_at].dist])

	# Self-intersection probe: any two samples far apart in arc length but close
	# in space and at a similar height would be a genuine crossing.
	var crossings := 0
	var worst := ""
	for i in path.samples.size():
		var a := path.samples[i]
		for j in range(i + 30, path.samples.size() - 10):
			var b := path.samples[j]
			if b.dist - a.dist < 60.0:
				continue
			var d := a.pos.distance_to(b.pos)
			if d < 18.0 and absf(a.pos.y - b.pos.y) < 6.0:
				crossings += 1
				if worst == "":
					worst = "samples %d(%.0fm, y%.1f) vs %d(%.0fm, y%.1f) dist %.1f" % [
						i, a.dist, a.pos.y, j, b.dist, b.pos.y, d]
	print("near-miss pairs: %d %s" % [crossings, worst])

	# Dump for plotting
	var f := FileAccess.open("res://dumps/centerline.txt", FileAccess.WRITE)
	f.store_line("# x y z width bank radial tangent_x tangent_y tangent_z")
	for s in path.samples:
		f.store_line("%.3f %.3f %.3f %.2f %.3f %d %.4f %.4f %.4f" % [
			s.pos.x, s.pos.y, s.pos.z, s.width, s.bank,
			1 if s.radial else 0, s.tangent.x, s.tangent.y, s.tangent.z])
	f.close()
	print("wrote res://dumps/centerline.txt")
